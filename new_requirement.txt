CONTROLLER

Create:

app/Http/Controllers/HealthcareHubController.php

METHOD 1

public function alberta()
{
$jobsQuery = Job::query()
->where(‘status’, ‘active’)
->where(‘state’, ‘Alberta’);

$jobCount = $jobsQuery->count();
$jobs = $jobsQuery
    ->latest()
    ->paginate(20);
$noIndex = $jobCount < 5;
return view('seo.healthcare-alberta', compact(
    'jobs',
    'jobCount',
    'noIndex'
));

}

METHOD 2

public function city($city)
{
$allowedCities = [
‘edmonton’ => ‘Edmonton’,
‘calgary’ => ‘Calgary’,
‘red-deer’ => ‘Red Deer’,
‘lethbridge’ => ‘Lethbridge’,
‘medicine-hat’ => ‘Medicine Hat’,
];

if (!array_key_exists($city, $allowedCities)) {
    abort(404);
}
$cityName = $allowedCities[$city];
$jobsQuery = Job::query()
    ->where('status', 'active')
    ->where('state', 'Alberta')
    ->where('city', $cityName);
$jobCount = $jobsQuery->count();
$jobs = $jobsQuery
    ->latest()
    ->paginate(20);
$noIndex = $jobCount < 5;
return view('seo.healthcare-city', compact(
    'city',
    'cityName',
    'jobs',
    'jobCount',
    'noIndex'
));

}
